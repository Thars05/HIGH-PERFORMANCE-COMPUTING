#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <sys/wait.h>
#include <unistd.h>
#include <time.h>
#include <sys/time.h>

int main()
{
    int n, i, j, k;

    printf("Enter matrix size (500 or 1000): ");

    if (scanf("%d", &n) != 1 || (n != 500 && n != 1000)) {
        printf("Invalid input. Please enter 500 or 1000.\n");
        return 1;
    }

    int size = n * n;


    int shmidA = shmget(IPC_PRIVATE, size * sizeof(int), IPC_CREAT | 0666);
    int shmidB = shmget(IPC_PRIVATE, size * sizeof(int), IPC_CREAT | 0666);
    int shmidC = shmget(IPC_PRIVATE, size * sizeof(int), IPC_CREAT | 0666);

    if (shmidA == -1 || shmidB == -1 || shmidC == -1) {
        perror("shmget failed");
        exit(1);
    }

    int *A = shmat(shmidA, NULL, 0);
    int *B = shmat(shmidB, NULL, 0);
    int *C = shmat(shmidC, NULL, 0);

    if (A == (int*)-1 || B == (int*)-1 || C == (int*)-1) {
        perror("shmat failed");
        exit(1);
    }

    srand(time(NULL));

    for (i = 0; i < size; i++) {
        A[i] = rand() % 10;
        B[i] = rand() % 10;
        C[i] = 0;
    }


    struct timeval start, end;
    gettimeofday(&start, NULL);


    for (i = 0; i < n; i++) {
        if (fork() == 0) {

            for (j = 0; j < n; j++) {
                int sum = 0;
                for (k = 0; k < n; k++) {
                    sum += A[i*n + k] * B[k*n + j];
                }
                C[i*n + j] = sum;
            }

            exit(0);
        }
    }


    for (i = 0; i < n; i++)
        wait(NULL);

    gettimeofday(&end, NULL);


    double time_taken = (end.tv_sec - start.tv_sec) * 1e3;
    time_taken += (end.tv_usec - start.tv_usec) * 1e-3;

    printf("\nParallel Fork Execution Time for %dx%d = %f milliseconds\n", n, n, time_taken);


    shmdt(A); shmdt(B); shmdt(C);
    shmctl(shmidA, IPC_RMID, NULL);
    shmctl(shmidB, IPC_RMID, NULL);
    shmctl(shmidC, IPC_RMID, NULL);

    return 0;
}
