#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main()
{
    int n, i, j, k;

    printf("Enter matrix size (500 or 1000): ");
    scanf("%d", &n);

    int *A = malloc(n * n * sizeof(int));
    int *B = malloc(n * n * sizeof(int));
    int *C = malloc(n * n * sizeof(int));

    if (!A || !B || !C) {
        printf("Memory allocation failed!\n");
        return 1;
    }

    srand(time(NULL));


    for (i = 0; i < n*n; i++) {
        A[i] = rand() % 10;
        B[i] = rand() % 10;
        C[i] = 0;
    }

    clock_t start = clock();


    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            int sum = 0;
            for (k = 0; k < n; k++) {
                sum += A[i*n + k] * B[k*n + j];
            }
            C[i*n + j] = sum;
        }
    }

    clock_t end = clock();

    double time = (double)(end - start) / CLOCKS_PER_SEC;
    printf("\nSerial Execution Time for %dx%d = %f seconds\n", n, n, time);

    free(A);
    free(B);
    free(C);

    return 0;
}
