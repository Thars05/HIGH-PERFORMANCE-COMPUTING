#include <stdio.h>
#include <omp.h>
#include <limits.h>

int main() {
    int n, i;
    printf("Enter size: ");
    scanf("%d", &n);
    int arr[n], res[n];
    printf("Enter elements: ");
    for(i = 0; i < n; i++) scanf("%d", &arr[i]);
    int num_threads = omp_get_max_threads();
    int chunk_min[num_threads];
    #pragma omp parallel for
    for (int t = 0; t < num_threads; t++) {
        int start = t * n / num_threads;
        int end = (t + 1) * n / num_threads;
        int local_min = INT_MAX;

        for (int j = start; j < end; j++) {
            if (arr[j] < local_min) local_min = arr[j];
        }
        chunk_min[t] = local_min;
    }
    for (i = 1; i < num_threads; i++) {
        if (chunk_min[i-1] < chunk_min[i]) chunk_min[i] = chunk_min[i-1];
    }
    #pragma omp parallel for
    for (int t = 0; t < num_threads; t++) {
        int start = t * n / num_threads;
        int end = (t + 1) * n / num_threads;
        int current_min = (t == 0) ? arr[0] : chunk_min[t-1];

        for (int j = start; j < end; j++) {
            if (arr[j] < current_min) current_min = arr[j];
            res[j] = current_min;
        }
    }

    printf("Result: ");
    for(i = 0; i < n; i++) printf("%d ", res[i]);
    printf("\n");

    return 0;
}
