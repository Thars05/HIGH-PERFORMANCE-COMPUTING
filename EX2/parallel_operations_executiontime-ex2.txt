#include <stdio.h>
#include <stdlib.h>
#include <time.h>


void print_Matrix(int **mat, int m, int n, const char *name) {
    int i,j;
    printf("\nMatrix %s:\n", name);
    for (i = 0; i < m; i++) {
        for (j = 0; j < n; j++) {
            printf("%d\t", mat[i][j]);
        }
        printf("\n");
    }
}


double get_Det(int **mat, int n) {
    double det = 1;
    int i, j, k;


    double **temp = (double **)malloc(n * sizeof(double *));
    for(i=0; i<n; i++){
        temp[i] = (double *)malloc(n * sizeof(double));
        for(j=0; j<n; j++){
            temp[i][j] = (double)mat[i][j];
        }
    }


    for (i = 0; i < n; i++) {

        if (temp[i][i] == 0) {

        }

        for (j = i + 1; j < n; j++) {
            double factor = temp[j][i] / temp[i][i];
            for (k = i; k < n; k++) {
                temp[j][k] -= factor * temp[i][k];
            }
        }
        det *= temp[i][i];
    }


    for(i=0; i<n; i++) free(temp[i]);
    free(temp);

    return det;
}

int main() {
    int m1, n1, m2, n2;
    int i, j, k;


    printf("\nEnter matrix A size [rows cols]: ");
    scanf("%d %d", &m1, &n1);
    printf("Enter matrix B size [rows cols]: ");
    scanf("%d %d", &m2, &n2);


    int **A = (int **)malloc(m1 * sizeof(int *));
    int **B = (int **)malloc(m2 * sizeof(int *));

    int **C = (int **)malloc(m1 * sizeof(int *));
    for (i = 0; i < m1; i++) A[i] = (int *)malloc(n1 * sizeof(int));
    for (i = 0; i < m2; i++) B[i] = (int *)malloc(n2 * sizeof(int));


    int max_col = (n1 > n2) ? n1 : n2;
    for (i = 0; i < m1; i++) C[i] = (int *)malloc(max_col * sizeof(int));


    printf("\nEnter elements for Matrix A (%dx%d):\n", m1, n1);
    for (i = 0; i < m1; i++) {
        for (j = 0; j < n1; j++) {
            printf("A[%d][%d]: ", i, j);
            scanf("%d", &A[i][j]);
        }
    }

    printf("\nEnter elements for Matrix B (%dx%d):\n", m2, n2);
    for (i = 0; i < m2; i++) {
        for (j = 0; j < n2; j++) {
            printf("B[%d][%d]: ", i, j);
            scanf("%d", &B[i][j]);
        }
    }

    struct timespec start, end;
    clock_gettime(CLOCK_MONOTONIC, &start);


    if (m1 == m2 && n1 == n2) {
        for (i = 0; i < m1; i++) {
            for (j = 0; j < n1; j++) {
                C[i][j] = A[i][j] + B[i][j];
            }
        }

        print_Matrix(C, m1, n1, "Addition (A+B)");
    } else {
        printf("\n[Addition Skipped]: Sizes do not match.\n");
    }


    if (m1 == m2 && n1 == n2) {
        for (i = 0; i < m1; i++) {
            for (j = 0; j < n1; j++) {
                C[i][j] = A[i][j] - B[i][j];
            }
        }
        print_Matrix(C, m1, n1, "Subtraction (A-B)");
    } else {
        printf("\n[Subtraction Skipped]: Sizes do not match.\n");
    }


    if (n1 == m2) {

        for (i = 0; i < m1; i++) {

            for (j = 0; j < n2; j++) {
                C[i][j] = 0;
                for (k = 0; k < n1; k++) {
                    C[i][j] += A[i][k] * B[k][j];
                }
            }
        }
        print_Matrix(C, m1, n2, "Multiplication (AxB)");
    } else {
        printf("\n[Multiplication Skipped]: Cols of A != Rows of B.\n");
    }


    if (m1 == n1) {
        double detA = get_Det(A, m1);
        printf("\nDeterminant of A: %.2f\n", detA);
    } else {
        printf("\n[Det A Skipped]: Not square.\n");
    }

    if (m2 == n2) {
        double detB = get_Det(B, m2);
        printf("\nDeterminant of B: %.2f\n", detB);
    } else {
        printf("\n[Det B Skipped]: Not square.\n");
    }

    clock_gettime(CLOCK_MONOTONIC, &end);
    double time_taken = (end.tv_sec - start.tv_sec) +
                        (end.tv_nsec - start.tv_nsec) / 1e9;

    printf("\n------------------------------\n");
    printf("Calculations took: %f seconds\n", time_taken);


    for(i=0; i<m1; i++) free(A[i]);
    for(i=0; i<m2; i++) free(B[i]);
    for(i=0; i<m1; i++) free(C[i]);
    free(A); free(B); free(C);

    return 0;
}
